% Water overlays do not mutate chemical snapshots and work in all formats.
\pgfplotsgetpourbaix{Fe1E-2}\waterbefore
\pgfplotsset{pourbaix/water/prepare={diagram=Fe1E-2}}
\pgfkeysgetvalue{/pgfplots/pourbaix/water/data/hydrogen b}\waterintercept
\ifdim\waterintercept pt=0pt\else\errmessage{Incorrect hydrogen intercept}\fi
\begingroup
\pgfplotsset{pourbaix/water/prepare={temperature=298.15,hydrogen pressure=100}}
\pgfkeysgetvalue{/pgfplots/pourbaix/water/data/hydrogen b}\waterchanged
\ifdim\waterchanged pt<-.059pt\else\errmessage{Pressure correction has wrong sign}\fi
\endgroup
\pgfkeysgetvalue{/pgfplots/pourbaix/water/data/hydrogen b}\waterrestored
\ifx\waterrestored\waterintercept\else\errmessage{Water preparation escaped scope}\fi
\tikzpicture
\axis[pourbaix/axis,width=7cm,height=5cm,xmin=0,xmax=14,ymin=-1,ymax=1.5]
\pgfplotsaddpourbaixwater[domain fills=true,
 domain/O2={label={$\hbox{O}_2$},label style={draw,fill=white}},
 domain/H2O={style={fill=cyan,fill opacity=.1}}];
\endaxis
\endtikzpicture
\pgfplotsgetpourbaix{Fe1E-2}\waterafter
\ifx\waterbefore\waterafter\else\errmessage{Water overlay changed the iron handle}\fi
\begingroup
\directlua{luapourbaix_water_saved_error=tex.error;tex.error=function() end}
\pgfplotsset{pourbaix/water/prepare={temperature=310}}
\pgfkeysgetvalue{/pgfplots/pourbaix/water/data/hydrogen b}\waterfailed
\ifx\waterfailed\waterintercept\else\errmessage{Failed preparation replaced water data}\fi
\pgfplotsset{pourbaix/water/prepare={diagram=Fe1E-2,temperature=310,oxygen standard potential=1.2}}
\pgfkeysgetvalue{/pgfplots/pourbaix/water/data/temperature}\waterfailedtemperature
\ifdim\waterfailedtemperature pt=298.15pt\else\errmessage{Temperature mismatch was committed}\fi
\directlua{tex.error=luapourbaix_water_saved_error;luapourbaix_water_saved_error=nil}
\def\pgferror#1{}%
\pgfplotsset{pourbaix/water/prepare={temperature=310,oxygen standard potential=1.2,hydrogen pressure}}
\pgfkeysgetvalue{/pgfplots/pourbaix/water/data/temperature}\waterfailedtemperature
\ifdim\waterfailedtemperature pt=298.15pt\else\errmessage{Missing water key value was committed}\fi
\pgfplotsset{pourbaix/water/prepare={temperature=310,oxygen standard potential=1.2,unknown input=1}}
\pgfkeysgetvalue{/pgfplots/pourbaix/water/data/temperature}\waterfailedtemperature
\ifdim\waterfailedtemperature pt=298.15pt\else\errmessage{Unknown water key was committed}\fi
\endgroup
% Preparation inside a plotting call is local, and its styles survive axis end.
\tikzpicture
\axis[pourbaix/axis,width=7cm,height=5cm,xmin=0,xmax=14,ymin=-1,ymax=1.5]
\pgfplotsaddpourbaixwater[prepare={temperature=298.15,hydrogen pressure=100},
 domain labels=false,boundary/H2={style={red},label={$h$},label style={above}}];
\pgfplotsaddpourbaixwater[domain labels=false,boundary style={green,dotted}];
\endaxis
\endtikzpicture

% Error recovery must never render with old water data or partial plot options.
\begingroup
\directlua{
 luapourbaixAudit = {water=luapourbaix.plot_water, element=luapourbaix.plot_diagram,
   error=tex.error, errors=0}
 luapourbaix.plot_water=function() error("Invalid water options reached rendering") end
 luapourbaix.plot_diagram=function() error("Invalid element options reached rendering") end
 tex.error=function() luapourbaixAudit.errors=luapourbaixAudit.errors+1 end
}
\def\pgferror#1{\directlua{luapourbaixAudit.errors=luapourbaixAudit.errors+1}}
\tikzpicture
\axis[xmin=0,xmax=14,ymin=-1,ymax=1.5]
\addpourbaixwater[prepare={temperature=310}];
\addpourbaixwater[prepare={temperature=298.15,unknown input=1}];
\addpourbaixwater[domain fills=invalid];
\addpourbaixwater[unknown option=1];
\addpourbaix[domain palette=invalid]{RenderFe};
\addpourbaix[unknown option=1]{RenderFe};
\addplot coordinates {(0,0) (14,0)};
\endaxis
\endtikzpicture
\directlua{
 assert(luapourbaixAudit.errors==6,tostring(luapourbaixAudit.errors))
 luapourbaix.plot_water=luapourbaixAudit.water
 luapourbaix.plot_diagram=luapourbaixAudit.element
 tex.error=luapourbaixAudit.error
 luapourbaixAudit=nil
}
\endgroup
