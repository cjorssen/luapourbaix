% Missing terminators fail before rendering and leave the next command intact.
\begingroup
\def\MissingTerminatorCount{0}
\def\ExpectedTerminatorError{luapourbaix: missing semicolon after Pourbaix plot command}
\def\pgferror#1{%
 \def\ActualTerminatorError{#1}%
 \ifx\ActualTerminatorError\ExpectedTerminatorError\else
  \errmessage{Incorrect missing-terminator diagnostic}%
 \fi
 \edef\MissingTerminatorCount{\number\numexpr\MissingTerminatorCount+1\relax}%
}
\addpourbaix{RenderFe}
\def\AfterMissingTerminator{preserved}
\addpourbaixwater[prepare={temperature=298.15}]
\relax
\pgfplotsaddpourbaix{RenderFe}
\relax
\pgfplotsaddpourbaixwater
\relax
\ifnum\MissingTerminatorCount=4\else\errmessage{Missing terminator was accepted}\fi
\def\ExpectedFollowingInput{preserved}
\ifx\AfterMissingTerminator\ExpectedFollowingInput\else
 \errmessage{Missing terminator consumed following input}\fi
\endgroup
