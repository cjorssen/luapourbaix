\pgfplotsset{pourbaix/declare={name=GalleryCu,element=Cu,species={Cu(s),Cu^2+,Cu2O(s),Cu(OH)2(s)},
 temperature=298.15,concentration=.01,boundary convention=equal element concentrations,constants={}},
 pourbaix/check=GalleryCu,pourbaix/calculate=GalleryCu,pourbaix/pH min=0,pourbaix/pH max=14,pourbaix/finalize=GalleryCu}
\tikzpicture\axis[pourbaix/axis,width=7cm,height=5cm,xmin=0,xmax=14,ymin=-0.8,ymax=1.5]
 \pgfplotsaddpourbaix{GalleryCu};
 \pgfplotsaddpourbaixwater[prepare={diagram=GalleryCu},domain labels=false];
\endaxis\endtikzpicture\par
