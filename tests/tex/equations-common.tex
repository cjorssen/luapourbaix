% Result transport, deferred plotting, grouping, idempotence and atomic failure.
\begingroup
\pgfplotsset{pourbaix/declare={test iron,name=Equations},pourbaix/check=Equations}%
\pgfplotsgetpourbaix{Equations}\beforeequations
\pgfplotsset{pourbaix/calculate=Equations}%
\setbox0=\hbox{\pgfplotsset{pourbaix/calculate=Equations}}%
\ifdim\wd0=0pt\else\errmessage{Equation export produced glue}\fi
\pgfplotsgetpourbaix{Equations}\equationhandle
\pgfkeys{/pgfplots/pourbaix/data/\equationhandle/a-temp/pH/.get=\verticalph}%
\pgfkeys{/pgfplots/pourbaix/data/\equationhandle/1-temp/expression/.get=\potentialexpression}%
\directlua{
  assert(math.abs(tonumber(token.get_macro("verticalph"))-7.5)<1e-10)
  assert(not luapourbaix.get(token.get_macro("beforeequations")).calculation)
}%
\pgfplotsset{pourbaix/calculate=Equations}%
\ifx\equationhandle\pgfplotslastpourbaix\else\errmessage{Calculation not idempotent}\fi
\par\tikzpicture
\axis[pourbaix/axis,width=7cm,height=5cm,domain=0:14,ymin=-1,ymax=1]
  \addplot[blue,samples=2] {\potentialexpression};
  \addplot[red] coordinates {(\verticalph,-1) (\verticalph,1)};
\endaxis
\endtikzpicture\par
\pgfplotsset{pourbaix/declare={test iron,name=Equations}}%
\let\uncalculated\pgfplotslastpourbaix
\pgfplotsset{pourbaix/calculate handle/.expanded=\equationhandle}%
\pgfplotsgetpourbaix{Equations}\aliasafterhandle
\ifx\uncalculated\aliasafterhandle\else\errmessage{Handle calculation changed alias}\fi
\directlua{
  luapourbaixEquationTest={original=tex.error,count=0}
  tex.error=function(message)
    assert(message:find("checked diagram",1,true))
    luapourbaixEquationTest.count=luapourbaixEquationTest.count+1
  end
}%
\pgfplotsset{pourbaix/calculate=Equations}%
\directlua{
  tex.error=luapourbaixEquationTest.original
  assert(luapourbaixEquationTest.count==1)
  luapourbaixEquationTest=nil
}%
\ifx\equationhandle\pgfplotslastpourbaix\else\errmessage{Failed calculation changed last handle}\fi
\endgroup
