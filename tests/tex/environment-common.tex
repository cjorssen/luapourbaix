\pgfplotsset{pourbaix/pH min=0,pourbaix/pH max=14,
 pourbaix/build={name=BuildTest,element=Fe,species={Fe(s),Fe^2+},
 temperature=298.15,concentration=.01,boundary convention=equal concentrations,
 thermodynamic database=none,
 constants={standard potential/Fe^2+|Fe(s)={value=-.44}}}}
\pgfplotsgetpourbaix{BuildTest}\BuildHandle
\pgfkeys{/pgfplots/pourbaix/data/\BuildHandle/final/1/b/.get=\BuildIntercept}

\directlua{luapourbaixBuildTest={original=tex.error,count=0};tex.error=function(message)
 assert(message:find('luapourbaix',1,true));luapourbaixBuildTest.count=luapourbaixBuildTest.count+1 end}

% Scalar transport and water inheritance share the same approximation.
\begingroup
\pgfplotsset{pourbaix/pH min=0,pourbaix/pH max=14,
 pourbaix/build={name=RoundedFe,element=Fe,species={Fe(s),Fe^2+},
 temperature=298.15,nernst factor=.06,concentration=.01,
 boundary convention=equal concentrations,constants={}},
 pourbaix/water/prepare={diagram=RoundedFe}}
\pgfplotsgetpourbaix{RoundedFe}\RoundedHandle
\pgfkeys{/pgfplots/pourbaix/data/\RoundedHandle/.cd,
 nernst factor/.get=\RoundedFactor,final/1/b/.get=\RoundedIntercept}
\pgfkeys{/pgfplots/pourbaix/water/data/a/.get=\RoundedWater}
\directlua{
 assert(tonumber(token.get_macro('RoundedFactor'))==.06)
 assert(math.abs(tonumber(token.get_macro('RoundedIntercept'))+.5)<1e-12)
 assert(tonumber(token.get_macro('RoundedWater'))==-.06)
 luapourbaixNernstTest={original=tex.error,count=0}
 tex.error=function(message)
  assert(message:find('Nernst',1,true))
  luapourbaixNernstTest.count=luapourbaixNernstTest.count+1 end
}
\pgfplotsset{pourbaix/water/prepare={diagram=RoundedFe,nernst factor=auto}}
\pgfplotsset{pourbaix/build={name=RoundedFe,element=Fe,species={Fe(s),Fe^2+},
 temperature=298.15,nernst factor=0,concentration=.01,
 boundary convention=equal concentrations,constants={}}}
\directlua{tex.error=luapourbaixNernstTest.original;assert(luapourbaixNernstTest.count==2);luapourbaixNernstTest=nil}
\ifx\RoundedHandle\pgfplotslastpourbaix\else\errmessage{Invalid factor changed diagram}\fi
\pgfkeys{/pgfplots/pourbaix/water/data/a/.get=\RoundedAfter}
\ifx\RoundedWater\RoundedAfter\else\errmessage{Invalid factor changed water}\fi
\endgroup
