% Literal pipe endpoints and deferred boundary nodes in all three formats.
\edef\AddressPipeBefore{\meaning|}
\pgfplotsset{pourbaix/declare={name=AddressFe,element=Fe,
 species={Fe(s),Fe^2+,Fe^3+,Fe(OH)2(s),Fe(OH)3(s)},
 species/Fe(OH)3(s)={thermodynamic phase=amorphous,oxidation number=3},
 temperature=298.15,concentration=.01,boundary convention=equal concentrations,
 constants={
  standard potential/Fe^3+|Fe^2+={value=.77},
  standard potential/Fe^2+|Fe(s)={value=-.44},
  solubility product/Fe(OH)2(s)={log value=-15},
  solubility product/Fe(OH)3(s)={value=1e-38}}},
 pourbaix/check=AddressFe,pourbaix/calculate=AddressFe,
 pourbaix/pH min=0,pourbaix/pH max=14,pourbaix/finalize=AddressFe}
\pgfplotsgetpourbaix{AddressFe}\AddressBefore
\directlua{
 local d=luapourbaix.get(token.get_macro('AddressBefore'))
 assert(d.constants[1].oxidized=='Fe^3+' and d.constants[1].reduced=='Fe^2+')
 assert(d.constants[3].solid=='Fe(OH)2(s)')
 assert(math.abs(d.boundaries_by_name['1-temp'].equation.b-(-.44-.059159349684781))<1e-12)
}
\par\tikzpicture\axis[xmin=0,xmax=14,ymin=-1.2,ymax=1.2,width=9cm,height=6cm]
 \pgfplotsaddpourbaix[
  boundary/1={style={red,thick},label=$r_1$,
    label style={above,pos=.65,name=addressed-redox}},
  boundary/a={style={dashed},label=$v$,label style={right,name=addressed-vertical}}
 ]{AddressFe};
\endaxis
\ifcsname pgf@sh@ns@addressed-redox\endcsname\else\errmessage{Missing addressed redox node}\fi
\ifcsname pgf@sh@ns@addressed-vertical\endcsname\else\errmessage{Missing addressed vertical node}\fi
\endtikzpicture\par
\pgfplotsgetpourbaix{AddressFe}\AddressAfter
\ifx\AddressBefore\AddressAfter\else\errmessage{Boundary presentation changed alias}\fi
\begingroup
