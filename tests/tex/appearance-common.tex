% Per-plot text/options survive deferred visualization and preserve snapshots.
\usetikzlibrary{patterns}
\pgfplotsset{pourbaix/declare={name=AppearanceFe,element=Fe,
 species={Fe(s),Fe^2+,Fe^3+,Fe(OH)2(s),Fe(OH)3(s)},
 temperature=298.15,concentration=.01,boundary convention=equal concentrations,
 constants={}},pourbaix/check=AppearanceFe,pourbaix/calculate=AppearanceFe,
 pourbaix/pH min=0,pourbaix/pH max=14,pourbaix/finalize=AppearanceFe}
\pgfplotsgetpourbaix{AppearanceFe}\AppearanceBefore
\def\AppearanceText{$Q$}
\par\tikzpicture
\axis[xmin=0,xmax=14,ymin=-1.2,ymax=1.2,width=9cm,height=6cm]
 \pgfplotsaddpourbaix[domain colors={yellow!20,blue!20,red!20,green!20},
   letter species={Fe^2+,Fe^3+},
   domain style={fill opacity=.4},
   domain/Fe(s)={label={\AppearanceText},label at={3,-.8},
     label style={draw,fill=white,name=appearance-label},style={pattern=north east lines}},
   domain/Fe^2+={label=$A$,label at={5,0},label style={text=blue,anchor=west}}
 ]{AppearanceFe};
\endaxis
\ifcsname pgf@sh@ns@appearance-label\endcsname\else
 \errmessage{Custom domain label node was not created}\fi
\endtikzpicture\par
\pgfplotspourbaix[xmin=0,xmax=14,ymin=-1.2,ymax=1.2,width=9cm,height=6cm,
 pourbaix/plot/domain coloring=none,pourbaix/plot/domain label mode=letters,
 pourbaix/plot/domain={species=Fe(s),label={$M$}}]{AppearanceFe}
\par
\pgfplotsgetpourbaix{AppearanceFe}\AppearanceAfter
\ifx\AppearanceBefore\AppearanceAfter\else\errmessage{Appearance changed snapshot}\fi
% Dynamic keys require a value and cannot redirect the species through data.
\begingroup
\def\AppearanceError{0}
\def\pgferror#1{\def\AppearanceError{1}}
\pgfplotsset{pourbaix/plot/domain/Fe(s)}
\ifnum\AppearanceError=1\else\errmessage{Missing domain value accepted}\fi
\def\AppearanceError{0}
\pgfplotsset{pourbaix/plot/domain/Fe(s)={species=Fe^2+,label=$X$}}
\ifnum\AppearanceError=1\else\errmessage{Domain path override accepted}\fi
\endgroup
% All named palettes work through ordinary deferred plots in every format.
\foreach \PaletteName in {tab10,tab10 light,grayscale,patterns,hatching,monochrome}{%
 \par\tikzpicture
 \axis[xmin=0,xmax=14,ymin=-1.2,ymax=1.2,width=8cm,height=5cm]
 \addpourbaix[domain palette/.expanded=\PaletteName,
   final domain label style={fill=white,fill opacity=.8,text opacity=1}]{AppearanceFe};
 \endaxis\endtikzpicture\par}
% The later plot must not change the earlier plot's captured pattern list.
\tikzpicture
\axis[xmin=0,xmax=14,ymin=-1.2,ymax=1.2,width=8cm,height=5cm]
 \addpourbaix[domain palette=patterns,domain style={pattern color=blue,fill opacity=.3}]{AppearanceFe};
 \addpourbaix[domain palette=tab10 light,domain style={fill opacity=.2}]{AppearanceFe};
\endaxis\endtikzpicture
\pgfplotsgetpourbaix{AppearanceFe}\AppearanceAfter
\ifx\AppearanceBefore\AppearanceAfter\else\errmessage{Palette changed snapshot}\fi
